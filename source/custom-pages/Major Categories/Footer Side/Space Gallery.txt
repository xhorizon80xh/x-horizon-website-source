<script>
(function () {

  "use strict";


  /* ============================================================
     ROOT
     ============================================================ */

  var root = document.getElementById("xh-space-gallery");

  if (!root) {
    return;
  }


  /* ============================================================
     ISOLATED SEARCH
     ============================================================ */

  var search =
    root.querySelector("#xh-gallery-search");

  var input =
    root.querySelector("#xh-gallery-search-input");

  var clear =
    root.querySelector("#xh-gallery-clear");

  var message =
    root.querySelector("#xh-gallery-search-message");


  if (!search || !input || !clear) {
    return;
  }


  /* ============================================================
     CLEAR BUTTON
     ============================================================ */

  clear.addEventListener("click", function () {

    input.value = "";

    input.focus();

    if (message) {
      message.style.display = "none";
      message.textContent = "";
    }

  });


  /* ============================================================
     SEARCH

     The hero itself does not redirect users automatically.

     This keeps the search completely independent and safe
     inside Odoo.

     Replace the search destination below if your gallery
     search system has a dedicated URL.
     ============================================================ */

  search.addEventListener("submit", function (event) {

    event.preventDefault();

    var query =
      input.value.trim();

    if (!query) {
      input.focus();
      return;
    }


    /*
      OPTIONAL SEARCH DESTINATION

      Example:

      window.location.href =
        "/space-gallery?search=" +
        encodeURIComponent(query);

      It is intentionally disabled here so the hero does not
      interfere with an existing Odoo search system.
    */


    if (message) {

      message.textContent =
        "Searching the Space Gallery for: " + query;

      message.style.display = "block";

    }

  });


  /* ============================================================
     ESCAPE KEY
     ============================================================ */

  input.addEventListener("keydown", function (event) {

    if (event.key === "Escape") {

      input.value = "";

      if (message) {
        message.style.display = "none";
        message.textContent = "";
      }

    }

  });


  /* ============================================================
     PREVENT EXTERNAL FORM BEHAVIOUR
     ============================================================ */

  search.setAttribute(
    "novalidate",
    "novalidate"
  );

})();
</script>
