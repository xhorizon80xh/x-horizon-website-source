    async function sha256(value){
      if(!window.crypto || !window.crypto.subtle || !window.TextEncoder) return '';
      const bytes = new TextEncoder().encode(String(value||'').trim());
      const hash = await window.crypto.subtle.digest('SHA-256', bytes);
      return Array.from(new Uint8Array(hash)).map(b => b.toString(16).padStart(2,'0')).join('');
    }
    async function isAdminKey(value, onUnlock){
      const normalized = String(value || '').trim();
      if(!normalized) return false;

      if(window.XHorizonAdmin && typeof window.XHorizonAdmin.unlock === 'function'){
        let unlockedByCallback = false;
        try{
          const ok = await Promise.resolve(window.XHorizonAdmin.unlock(normalized, function(){
            unlockedByCallback = true;
            if(typeof onUnlock === 'function') onUnlock();
          }));
          if(ok && !unlockedByCallback && typeof onUnlock === 'function') onUnlock();
          return !!ok || unlockedByCallback;
        }catch(error){
          console.error('X Horizon admin unlock failed:', error);
          return false;
        }
      }

      const directKeys = runtimeAdminKeys();
      const normalKey = normalizeKey(normalized);
      if(directKeys.length && directKeys.includes(normalKey)){
        if(typeof onUnlock === 'function') onUnlock();
        return true;
      }
      const hashedKeys = runtimeAdminHashes();
      if(!hashedKeys.length) return false;
      const digest = await sha256(normalized);
      const ok = !!digest && hashedKeys.includes(digest);
      if(ok && typeof onUnlock === 'function') onUnlock();
      return ok;
    }

      } else if(el.src !== FALLBACK){
        el.src = FALLBACK;
        el.classList.add('xhuc-fallback-img');
      }
    }

          image: articleImage,

          blocks: []

    }


    function ensureArticleMeta(card){
      if(!card) return null;
      if(!articles[card.slug]) articles[card.slug] = createDefaultArticles([card])[card.slug];
      const article = articles[card.slug];
      const cardArticleImage = card.articleImage || card.image || FALLBACK;
      const cardArticleSource = card.articleSource || card.source || 'NASA / official source';
      article.slug = article.slug || card.slug;
      article.title = article.title || card.title;
      if(!article.customImage || !article.image || article.image === FALLBACK){
        article.image = cardArticleImage;
        article.source = cardArticleSource;
      }
      article.source = article.source || cardArticleSource;
      article.tags = article.tags || card.tags || '';
      article.blocks = Array.isArray(article.blocks) ? article.blocks : [];
      return article;
    }

    function nextCardId(){
      const nums = cards.map(c => parseInt(String(c.id).replace(/\D/g,''),10)).filter(Boolean);
      const next = (Math.max(0,...nums)+1).toString().padStart(3,'0');
      return 'GLS-' + next;
    }

    function saveCardFromForm(){
      const id = document.getElementById('xhuc-admin-id').value.trim() || nextCardId();
      const title = document.getElementById('xhuc-admin-title').value.trim();
      if(!title){ alert('Add a title first.'); return; }
      const card = {
        id,
        slug: slugify(title),
        title,
        desc: document.getElementById('xhuc-admin-desc').value.trim() || 'Add a clean concept description from card admin.',
        image: document.getElementById('xhuc-admin-img').value.trim() || FALLBACK,
        source: document.getElementById('xhuc-admin-source').value.trim() || 'Custom / official source',
        tags: document.getElementById('xhuc-admin-tags').value.trim()
      };
      const indexById = cards.findIndex(c => c.id === id);
      const indexBySlug = cards.findIndex(c => c.slug === card.slug);
      const index = indexById >= 0 ? indexById : indexBySlug;
      if(index >= 0) cards[index] = card; else cards.push(card);
      if(!articles[card.slug]) articles[card.slug] = createDefaultArticles([card])[card.slug];
      save(STORAGE_CARDS,cards); save(STORAGE_ARTICLES,articles); renderCards(cardSearch.value); renderAdminSelect(); const sel=document.getElementById('xhuc-admin-select'); if(sel) sel.value=id;
    }

      save(STORAGE_CARDS,cards); save(STORAGE_ARTICLES,articles); fillCardForm(null); renderCards(cardSearch.value);
    }

    function bindArticleManager(){

      const infoBtn = document.getElementById('xhuc-save-article-info');

      if(deleteBtn) deleteBtn.addEventListener('click', deleteCurrentArticleContent);

    function setSaveState(text, mode='ok'){
      const state = document.getElementById('xhuc-save-state');
      if(!state) return;
      const stamp = new Date().toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'});
      state.textContent = `${text} • ${stamp}`;
      state.classList.toggle('xhuc-ok', mode !== 'warn');
      state.classList.toggle('xhuc-warn', mode === 'warn');
    }

    function renderArticleBlocks(){
      const canvas = document.getElementById('xhuc-article-canvas');
      const article = getArticle();
      if(!canvas || !article) return;
      let html = '';
      article.blocks.forEach((block, index) => {
        if(articleAdminUnlocked) html += dropzone(index);
        html += blockHTML(block, index);
      });
      if(articleAdminUnlocked) html += dropzone(article.blocks.length);
      canvas.innerHTML = html || '<div class="xhuc-blank-canvas" aria-label="Blank article canvas"></div>';
      bindDropzones();
      bindBlocks();
      applySelectedState();
    }

    function dropzone(index){return `<div class="xhuc-dropzone" data-drop-index="${index}">Drop content here</div>`}

    function blockHTML(block, index){
      const style = block.style || {};
      const size = parseInt(style.size || 16,10);
      const widthRaw = style.width === 'auto' ? 'auto' : parseInt(style.width || 100,10);
      const width = widthRaw === 'auto' ? 'auto' : Math.max(10, Math.min(100, widthRaw)) + '%';
      const height = Math.max(0, parseInt(style.height || 0,10));
      const align = ['left','center','right'].includes(style.align) ? style.align : 'left';
      const color = style.color || '#f4f7fb';
      const bg = style.bg || '#050505';
      const motion = style.motion && style.motion !== 'none' ? ' xhuc-motion-' + style.motion : '';
      const line = style.line ? ' xhuc-line-style' : '';
      const frameH = height ? height + 'px' : 'auto';
      const pad = Math.max(0, Math.min(90, parseInt(style.padding ?? 18,10))) + 'px';
      const radius = Math.max(0, Math.min(48, parseInt(style.radius ?? 18,10))) + 'px';
      const borderStyle = ['solid','dashed','dotted','none'].includes(style.borderStyle) ? style.borderStyle : 'solid';
      const weight = ['400','600','800','900'].includes(String(style.weight)) ? String(style.weight) : '400';
      const base = `draggable="${articleAdminUnlocked ? 'true' : 'false'}" data-block-id="${esc(block.id)}" data-block-index="${index}" data-block-type="${esc(block.type)}" data-align="${esc(align)}" style="--block-size:${size}px;--block-width:${width};--block-align:${esc(align)};--block-color:${esc(color)};--block-bg:${esc(bg)};--block-frame-height:${frameH};--block-img-height:${frameH};--block-pad:${pad};--block-radius:${radius};--block-border-style:${esc(borderStyle)};--block-weight:${weight}"`;
      const adminTools = articleAdminUnlocked ? `<span class="xhuc-block-drag-handle" title="Drag this block">Drag</span><div class="xhuc-block-actions"><button class="xhuc-block-action" type="button" data-block-edit="${esc(block.id)}">Edit</button><button class="xhuc-block-action" type="button" data-block-copy="${esc(block.id)}">Copy</button><button class="xhuc-block-action xhuc-danger" type="button" data-block-delete="${esc(block.id)}">Delete</button></div>` : '';
      const label = articleAdminUnlocked ? `<span class="xhuc-block-label">${esc(block.type)}</span>` : '';
      const emptyAdmin = msg => articleAdminUnlocked ? `<div class="xhuc-empty-block">${esc(msg)}</div>` : '';
      const content = String(block.content || '');

      if(block.type === 'heading') return `<section class="xhuc-block${motion}${line}" ${base}>${adminTools}${label}<h2 contenteditable="${articleAdminUnlocked}">${esc(content)}</h2></section>`;
      if(block.type === 'short') return `<section class="xhuc-block${motion}${line}" ${base}>${adminTools}${label}<p contenteditable="${articleAdminUnlocked}">${esc(content)}</p></section>`;
      if(block.type === 'long') return `<section class="xhuc-block${motion}${line}" ${base}>${adminTools}${label}<div class="xhuc-long-text" contenteditable="${articleAdminUnlocked}">${esc(content)}</div></section>`;
      if(block.type === 'tags'){
        const tags = content.split(',').map(t => t.trim()).filter(Boolean);
        const tagHTML = tags.length ? tags.map(t=>`<span>${esc(t)}</span>`).join('') : emptyAdmin('Paste comma-separated tags in Edit.');
        return `<section class="xhuc-block${motion}${line}" ${base}>${adminTools}${label}<div class="xhuc-taglist" contenteditable="${articleAdminUnlocked}">${tagHTML}</div></section>`;
      }
      if(block.type === 'picture'){
        const img = content ? `<img src="${esc(content)}" alt="Article image" loading="lazy" onerror="xhucImgError(this)">` : emptyAdmin('Paste image URL in Edit.');
        const cap = block.caption ? `<div class="xhuc-caption" contenteditable="${articleAdminUnlocked}">${esc(block.caption)}</div>` : (articleAdminUnlocked ? `<div class="xhuc-caption" contenteditable="${articleAdminUnlocked}">Add caption/source in Edit</div>` : '');
        return `<section class="xhuc-block xhuc-block-picture${motion}${line}" ${base}>${adminTools}${label}${img}${cap}</section>`;
      }
      if(block.type === 'equation'){
        const inner = content ? `<iframe class="xhuc-frame xhuc-equation-frame" sandbox="allow-scripts allow-same-origin allow-forms allow-popups" srcdoc="${esc(frameDoc(content))}"></iframe>` : emptyAdmin('Paste external equation code in Edit.');
        return `<section class="xhuc-block${motion}${line}" ${base}>${adminTools}${label}${inner}</section>`;
      }
      if(block.type === 'embed'){
        const inner = content ? `<iframe class="xhuc-frame" sandbox="allow-scripts allow-same-origin allow-forms allow-popups" srcdoc="${esc(frameDoc(content))}"></iframe>` : emptyAdmin('Paste simulation / iframe / embed code in Edit.');
        return `<section class="xhuc-block${motion}${line}" ${base}>${adminTools}${label}${inner}</section>`;
      }
      return `<section class="xhuc-block${motion}${line}" ${base}>${adminTools}${label}<div contenteditable="${articleAdminUnlocked}">${esc(content)}</div></section>`;
    }

    function frameDoc(code){
      return `<!doctype html><html><head><meta charset='utf-8'><meta name='viewport' content='width=device-width,initial-scale=1'><style>html,body{margin:0;min-height:100%;background:#05070b;color:#f8fbff;font-family:Inter,Arial,sans-serif}body{padding:18px;box-sizing:border-box}.xh-frame-note{border:1px solid rgba(255,255,255,.12);background:#090d15;border-radius:14px;padding:18px;line-height:1.5}</style></head><body>${code}</body></html>`;
    }

    function bindDropzones(){
      document.querySelectorAll('.xhuc-dropzone').forEach(zone => {
        zone.addEventListener('dragover', e => { e.preventDefault(); zone.classList.add('xhuc-over'); });
        zone.addEventListener('dragleave', () => zone.classList.remove('xhuc-over'));
        zone.addEventListener('drop', e => {
          e.preventDefault(); zone.classList.remove('xhuc-over');
          const index = parseInt(zone.dataset.dropIndex,10);
          const movingBlock = e.dataTransfer.getData('application/xhuc-block') || draggedBlockId;
          const type = e.dataTransfer.getData('application/xhuc-tool') || e.dataTransfer.getData('text/plain') || draggedTool;
          if(movingBlock){ moveBlockToIndex(movingBlock,index); }
          else if(type){ addBlock(type,index,true); }
          collapseEditorDuringDrag(false);
          draggedTool = null; draggedBlockId = null;
        });
      });
    }

    function bindBlocks(){
      document.querySelectorAll('.xhuc-block').forEach(blockEl => {
        blockEl.addEventListener('dragstart', e => {
          if(!articleAdminUnlocked) return;
          if(e.target.closest('input,textarea,select,button,[contenteditable="true"]')){ e.preventDefault(); return; }
          draggedBlockId = blockEl.dataset.blockId;
          draggedTool = null;
          e.dataTransfer.setData('application/xhuc-block', draggedBlockId);
          e.dataTransfer.setData('text/plain', draggedBlockId);
        });
        blockEl.addEventListener('dragend', () => { draggedBlockId = null; });
        blockEl.addEventListener('click', e => {
          if(!articleAdminUnlocked) return;
          const editBtn = e.target.closest('[data-block-edit]');
          const copyBtn = e.target.closest('[data-block-copy]');
          const delBtn = e.target.closest('[data-block-delete]');
          if(editBtn){ selectedBlockId = editBtn.dataset.blockEdit; applySelectedState(); showEditbar(); openInspector(); return; }
          if(copyBtn){ selectedBlockId = copyBtn.dataset.blockCopy; duplicateBlock(); return; }
          if(delBtn){ selectedBlockId = delBtn.dataset.blockDelete; deleteBlock(); return; }
          selectedBlockId = blockEl.dataset.blockId; showEditbar(); applySelectedState();
        });
        blockEl.addEventListener('dblclick', e => {
          if(!articleAdminUnlocked) return;
          selectedBlockId = blockEl.dataset.blockId; applySelectedState(); showEditbar(); openInspector();
        });
        blockEl.addEventListener('blur', e => {
          if(!articleAdminUnlocked) return;
          const block = getBlock(blockEl.dataset.blockId);
          if(!block) return;
          if(['heading','short','long'].includes(block.type)){
            const editable = blockEl.querySelector('[contenteditable="true"]');
            if(editable) block.content = editable.innerText.trim();
          }
          if(block.type === 'picture'){
            const cap = blockEl.querySelector('.xhuc-caption');
            if(cap) block.caption = cap.innerText.trim();
          }
          if(block.type === 'tags'){
            block.content = Array.from(blockEl.querySelectorAll('span')).map(s=>s.innerText.trim()).filter(Boolean).join(', ');
          }
          saveArticles();
        }, true);
      });
    }


    function captureInlineEdits(){
      document.querySelectorAll('.xhuc-block').forEach(blockEl => {
        const block = getBlock(blockEl.dataset.blockId);
        if(!block) return;
        if(['heading','short','long'].includes(block.type)){
          const editable = blockEl.querySelector('[contenteditable="true"]');
          if(editable) block.content = editable.innerText.trim();
        }
        if(block.type === 'picture'){
          const cap = blockEl.querySelector('.xhuc-caption');
          if(cap) block.caption = cap.innerText.trim();
        }
        if(block.type === 'tags'){
          block.content = Array.from(blockEl.querySelectorAll('span')).map(s=>s.innerText.trim()).filter(Boolean).join(', ');
        }
      });
    }

    function addBlock(type, index, openEditor=false){
      if(!articleAdminUnlocked){ alert('Unlock article editor from the article search bar first.'); return; }
      const article = getArticle();
      if(!article) return;
      const block = makeBlock(type);
      article.blocks.splice(index,0,block);
      selectedBlockId = block.id;
      saveArticles(); renderArticleBlocks(); showEditbar();
      if(openEditor) setTimeout(openInspector, 30);
    }

    function makeBlock(type){
      const base = {id:uid(type), type, content:'', caption:'', style:{size:16,width:100,height:0,align:'left',color:'#f4f7fb',bg:'#050505',motion:'none',line:false,padding:18,radius:18,borderStyle:'solid',weight:'400'}};
      if(type === 'heading'){base.style.size = 40;}
      if(type === 'short'){base.style.size = 20; base.style.color = '#cbd5e1';}
      if(type === 'long'){base.style.size = 16;}
      if(type === 'picture'){base.style.bg = '#050505'; base.style.height = 0;}
      if(type === 'tags'){base.style.size = 14;}
      if(type === 'embed'){base.style.bg = '#050505'; base.style.height = 0;}
      if(type === 'equation'){base.style.bg = '#050505'; base.style.height = 0;}
      return base;
    }

    function getBlock(id){
      const article = getArticle();
      if(!article) return null;
      return article.blocks.find(b => b.id === id);
    }
    function blockIndex(id){
      const article = getArticle();
      return article ? article.blocks.findIndex(b => b.id === id) : -1;

    function updateSelectedStyle(prop,value){

    }
    function moveBlock(dir){
      const article = getArticle(); const i = blockIndex(selectedBlockId); if(!article || i<0) return;
      const n = i + dir; if(n<0 || n>=article.blocks.length) return;
      const [b] = article.blocks.splice(i,1); article.blocks.splice(n,0,b); saveArticles(); renderArticleBlocks();
    }
    function moveBlockToIndex(blockId,index){
      const article = getArticle();
      if(!article) return;
      const oldIndex = article.blocks.findIndex(b => b.id === blockId);
      if(oldIndex < 0) return;
      const [block] = article.blocks.splice(oldIndex,1);
      let target = Math.max(0, Math.min(article.blocks.length, index));
      if(oldIndex < index) target = Math.max(0, target - 1);
      article.blocks.splice(target,0,block);
      selectedBlockId = block.id;
      saveArticles(); renderArticleBlocks(); showEditbar();
    }
    function duplicateBlock(){
      const article = getArticle(); const i = blockIndex(selectedBlockId); if(!article || i<0) return;
      const copy = JSON.parse(JSON.stringify(article.blocks[i])); copy.id = uid(copy.type); article.blocks.splice(i+1,0,copy); selectedBlockId = copy.id; saveArticles(); renderArticleBlocks(); showEditbar();
    }
    function deleteBlock(){
      const article = getArticle(); const i = blockIndex(selectedBlockId); if(!article || i<0) return;
      if(!confirm('Delete this article block?')) return;
      article.blocks.splice(i,1); selectedBlockId = null; saveArticles(); renderArticleBlocks(); editbar.classList.remove('xhuc-show'); closeInspector();
    }

    function openInspector(){
      const block = getBlock(selectedBlockId); if(!block || !inspector) return;
      document.getElementById('xhuc-inspector-title').textContent = 'Edit ' + block.type.toUpperCase();
      const text = document.getElementById('xhuc-inspector-text');
      const url = document.getElementById('xhuc-inspector-url');
      const caption = document.getElementById('xhuc-inspector-caption');
      const urlLabel = document.getElementById('xhuc-inspector-url-label');
      const captionLabel = document.getElementById('xhuc-inspector-caption-label');
      const textLabel = document.getElementById('xhuc-inspector-text-label');
      const w = document.getElementById('xhuc-inspector-width');
      const h = document.getElementById('xhuc-inspector-height');

      url.value = block.type === 'picture' ? (block.content || '') : '';
      caption.value = block.caption || '';
      text.value = block.type === 'picture' ? '' : (block.content || '');
      w.value = parseInt(block.style?.width || 100,10);
      h.value = parseInt(block.style?.height || 0,10);

      const isPicture = block.type === 'picture';
      const isCode = block.type === 'embed' || block.type === 'equation';
      url.style.display = isPicture ? 'block' : 'none';
      urlLabel.style.display = isPicture ? 'block' : 'none';
      caption.style.display = isPicture ? 'block' : 'none';
      captionLabel.style.display = isPicture ? 'block' : 'none';
      text.style.display = isPicture ? 'none' : 'block';
      textLabel.style.display = isPicture ? 'none' : 'block';
      if(block.type === 'heading') text.placeholder = 'Write heading text here';
      else if(block.type === 'short') text.placeholder = 'Write short description here';
      else if(block.type === 'long') text.placeholder = 'Write long article context here';
      else if(block.type === 'tags') text.placeholder = 'Paste tags separated by commas';
      else if(isCode) text.placeholder = 'Paste external HTML / CSS / JS / iframe / equation code here';
      else text.placeholder = 'Write content here';
      inspector.classList.add('xhuc-show');
      setTimeout(() => (isPicture ? url : text).focus(), 40);
    }
    function closeInspector(){ if(inspector) inspector.classList.remove('xhuc-show'); }
    function applyInspector(){
      const block = getBlock(selectedBlockId); if(!block) return;
      const text = document.getElementById('xhuc-inspector-text');

      block.style = block.style || {};

      block.style.height = Math.max(0, Math.min(1600, parseInt(h.value || 0,10)));
      saveArticles(); renderArticleBlocks(); showEditbar(); closeInspector();
    }
