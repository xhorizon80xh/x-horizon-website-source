    function requestRootAdminUnlock(value, onUnlock){
      const key = String(value || '').trim();
      if(!key) return false;

      function status(message){
        if(adminPill){
          adminPill.classList.remove('xhms-hidden');
          adminPill.textContent = message;
        }
      }

      const bridge =
        (window.XHorizonAdmin && typeof window.XHorizonAdmin.unlock === 'function' && window.XHorizonAdmin) ||
        (window.xHorizonAdmin && typeof window.xHorizonAdmin.unlock === 'function' && window.xHorizonAdmin) ||
        (window.XHorizonFirebaseConfigurator && typeof window.XHorizonFirebaseConfigurator.unlock === 'function' && window.XHorizonFirebaseConfigurator) ||
        null;

      if(!bridge){
        status('Admin root config missing');
        return false;
      }

      let unlockedByCallback = false;
      let settled = false;
      const finish = () => {
        if(settled) return;
        settled = true;
        unlockedByCallback = true;
        status('Admin unlocked');
        if(typeof onUnlock === 'function') onUnlock();
      };

      status('Checking admin key...');
      try{
        const result = bridge.unlock(key, finish);
        Promise.resolve(result).then(function(ok){
          if((ok === true || (ok && (ok.ok || ok.success || ok.unlocked))) && !unlockedByCallback) finish();
          if(ok === false && !unlockedByCallback) status('Wrong admin key');
        }).catch(function(err){
          console.error('X Horizon admin unlock failed:', err);
          if(!unlockedByCallback) status('Firebase admin check failed');
        });
      }catch(err){
        console.error('X Horizon admin unlock failed:', err);
        status('Firebase admin check failed');
      }
      return true;
    }

<script id="xhms-final-visual-theme-fix-script">
(function(){
  var heroUrl = 'https://images-assets.nasa.gov/image/iss063e026129/iss063e026129~large.jpg';
  function fixHero(){
    var page = document.getElementById('xhms-page');
    var hero = document.getElementById('xhms-hero');
    if(page) page.style.setProperty('--xhms-hero-img', "url('" + heroUrl + "')");
    if(hero && !hero.querySelector('.xhms-hero-bg-hard')){
      var img = document.createElement('img');
      img.className = 'xhms-hero-bg-hard';
      img.src = heroUrl;
      img.alt = '';
      img.setAttribute('aria-hidden','true');
      hero.insertBefore(img, hero.firstChild);
    }
    var bg = hero && hero.querySelector('.xhms-hero-bg-hard');
    if(bg){
      bg.onerror = function(){ this.onerror = null; this.src = 'https://images-assets.nasa.gov/image/as11-40-5874/as11-40-5874~large.jpg'; };
      if(!bg.src || bg.src.indexOf('iss063e026129') === -1) bg.src = heroUrl;
    }
    var btn = document.getElementById('xhms-clear-search');
    if(btn) btn.textContent = 'Search';
    var kicker = document.querySelector('#xhms-page .xhms-kicker');
    if(kicker) kicker.style.display = 'none';
  }
  if(document.readyState === 'loading') document.addEventListener('DOMContentLoaded', fixHero); else fixHero();
  window.addEventListener('load', fixHero);
  setTimeout(fixHero, 250);
})();
</script>
