<script type="module">
/* X HORIZON Firebase admin bridge: uses existing root bridge if present, otherwise initializes from safe global config. */
import { initializeApp, getApps } from "https://www.gstatic.com/firebasejs/10.12.5/firebase-app.js";
import { getAuth, signInWithCustomToken } from "https://www.gstatic.com/firebasejs/10.12.5/firebase-auth.js";
import { getFunctions, httpsCallable } from "https://www.gstatic.com/firebasejs/10.12.5/firebase-functions.js";

(function(){
  const cfg = window.XHORIZON_FIREBASE_CONFIG || window.XHorizonFirebaseConfig || null;
  const fnName = window.XHORIZON_ADMIN_VERIFY_FUNCTION || "verifyAdminKey";
  function configured(c){
    return !!(c && c.apiKey && c.projectId && !String(c.apiKey).includes("YOUR_") && !String(c.projectId).includes("YOUR_"));
  }
  if (window.XHorizonAdmin && typeof window.XHorizonAdmin.unlock === "function") {
    window.XHorizonFirebaseReady = true;
    return;
  }
  if (!configured(cfg)) {
    window.XHorizonFirebaseReady = false;
    window.XHorizonAdmin = window.XHorizonAdmin || {
      unlock: async function(){
        console.error("X Horizon Firebase config missing. Define window.XHORIZON_FIREBASE_CONFIG in root/Odoo General JS.");
        return false;
      }
    };
    return;
  }
  let app = null, auth = null, functions = null;
  try {
    app = getApps().length ? getApps()[0] : initializeApp(cfg);
    auth = getAuth(app);
    functions = getFunctions(app);
    window.XHorizonFirebaseReady = true;
  } catch (error) {
    console.error("X Horizon Firebase initialization failed:", error);
    window.XHorizonFirebaseReady = false;
    return;
  }
  window.XHorizonAdmin = {
    async unlock(key, done){
      const cleanKey = String(key || "").trim();
      if (!cleanKey || !auth || !functions) return false;
      try {
        const verifyAdminKey = httpsCallable(functions, fnName);
        const result = await verifyAdminKey({ key: cleanKey });
        const data = result && result.data ? result.data : {};
        const allowed = data.allowed === true || data.admin === true || data.success === true;
        const customToken = data.token || data.customToken || data.firebaseToken || "";
        if (customToken) await signInWithCustomToken(auth, customToken);
        let claimAllowed = false;
        if (auth.currentUser) {
          const tokenResult = await auth.currentUser.getIdTokenResult(true);
          claimAllowed = tokenResult.claims && (tokenResult.claims.admin === true || tokenResult.claims.xhAdmin === true || tokenResult.claims.role === "admin");
        }
        const ok = allowed || claimAllowed;
        if (ok && typeof done === "function") done();
        return ok;
      } catch (error) {
        console.error("X Horizon admin verification failed:", error);
        return false;
      }
    }
  };
})();

      if (searchInput) {
        searchInput.value = "";
        searchInput.placeholder = "Search concepts...";
      }
      setAdminStatus("");
      filterCards("");
    }

    function filterCards(query) {
      var normalized = normalizeText(query);
      var cards = getCards();
      var visibleCount = 0;

      cards.forEach(function (card) {
        var data = getCardData(card, 0);
        var haystack = normalizeText(data.title + " " + data.description + " " + data.tags.join(" "));
        var match = !normalized || haystack.indexOf(normalized) > -1;
        card.style.display = match ? "" : "none";
        if (match) visibleCount += 1;
      });

      if (normalized && visibleCount === 0) {
        addClass(page, "xh-empty-search");
      } else {
        page.className = page.className.replace(/\bxh-empty-search\b/g, "").replace(/\s+/g, " ").trim();
      }
    }

    loadLocalCards();
    renumberCards();

    function setSearchButtonState(isChecking) {
      if (!searchButton) return;
      searchButton.disabled = !!isChecking;
      searchButton.textContent = isChecking ? "Checking" : "Search";
    }

    function handleSearchSubmit() {
      var value = searchInput ? searchInput.value.trim() : "";
      if (!value) {
        filterCards("");
        return;
      }

      if (window.XHorizonAdmin && typeof window.XHorizonAdmin.unlock === "function") {
        setSearchButtonState(true);
        searchInput.placeholder = "Checking admin authorization...";
        setAdminStatus("Checking Firebase admin authorization...");

        var unlockedByCallback = false;
        Promise.resolve(window.XHorizonAdmin.unlock(value, function () {
          unlockedByCallback = true;
          openAdminMode();
        })).then(function (ok) {
          if (ok && !unlockedByCallback) {
            openAdminMode();
            return;
          }
          if (!ok) {
            setAdminStatus("No admin match. Showing search results.");
            searchInput.placeholder = "Search concepts...";
            filterCards(value);
          }
        }).catch(function (error) {
          console.error("X Horizon admin unlock failed:", error);
          setAdminStatus("Firebase admin authorization failed. Showing search results.");
          searchInput.placeholder = "Search concepts...";
          filterCards(value);
        }).finally(function () {
          setSearchButtonState(false);
        });
        return;
      }

      setAdminStatus("Firebase admin bridge not found. Showing search results.");
      filterCards(value);
    }

    if (searchInput) {
      searchInput.addEventListener("input", function () {
        filterCards(searchInput.value);
      });
    }

    if (searchForm) {
      searchForm.addEventListener("submit", function (event) {
        event.preventDefault();
        handleSearchSubmit();
      });
    }

    if (adminSelect) {
      adminSelect.addEventListener("change", loadAdminFields);
    }

    if (adminSave) {
      adminSave.addEventListener("click", function () {
        var index = parseInt(adminSelect && adminSelect.value ? adminSelect.value : "0", 10);
        var card = getCards()[index];
        applyDataToCard(card, readAdminFields());
        renumberCards();
        saveLocalCards();
        fillAdminSelect();
        if (adminSelect) adminSelect.value = String(index);
      });
    }

    if (adminAdd) {
      adminAdd.addEventListener("click", function () {
        if (!grid) return;
        var data = readAdminFields();
        grid.appendChild(makeCard(data, getCards().length));
        renumberCards();
        saveLocalCards();
        fillAdminSelect();
        if (adminSelect) adminSelect.value = String(getCards().length - 1);
        loadAdminFields();
      });
    }

    if (adminDelete) {
      adminDelete.addEventListener("click", function () {
        var index = parseInt(adminSelect && adminSelect.value ? adminSelect.value : "0", 10);
        var card = getCards()[index];
        if (card && card.parentNode) {
          card.parentNode.removeChild(card);
          renumberCards();
          saveLocalCards();
          fillAdminSelect();
          filterCards(searchInput ? searchInput.value : "");
        }
      });
    }

    if (adminReset) {
      adminReset.addEventListener("click", function () {
        try {
          window.localStorage.removeItem(storageKey);
        } catch (e) {}
        window.location.reload();
      });
    }

    if (adminLock) {
      adminLock.addEventListener("click", closeAdminMode);
    }

    var reduceMotion = false;
    try {
      reduceMotion = window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    } catch (e) {}

    var revealItems = Array.prototype.slice.call(page.querySelectorAll(".xh-reveal"));

    function revealOne(el) {
      if (!el || el.getAttribute("data-xh-shown") === "1") return;
      el.setAttribute("data-xh-shown", "1");
      addClass(el, "xh-visible");
    }

    if (reduceMotion) {
      revealItems.forEach(revealOne);
      return;
    }

    if ("IntersectionObserver" in window) {
      var observer = new IntersectionObserver(function (entries) {
        var visible = entries
          .filter(function (entry) {
            return entry.isIntersecting;
          })
          .sort(function (a, b) {
            return revealItems.indexOf(a.target) - revealItems.indexOf(b.target);
          });

        visible.forEach(function (entry, index) {
          window.setTimeout(function () {
            revealOne(entry.target);
            observer.unobserve(entry.target);
          }, index * 145);
        });
      }, {
        threshold: 0.18,
        rootMargin: "0px 0px -8% 0px"
      });

      revealItems.forEach(function (el) {
        observer.observe(el);
      });
    } else {
      revealItems.forEach(function (el, index) {
        window.setTimeout(function () {
          revealOne(el);
        }, index * 130);
      });
    }
  });
})();
</script>
