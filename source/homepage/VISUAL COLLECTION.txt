<script>
(function () {
  "use strict";

  // Scoped to this block only: no global ids, no document-wide listeners.
  var roots = document.querySelectorAll("[data-xhc-root]");

  Array.prototype.forEach.call(roots, function (root) {
    if (root.getAttribute("data-xhc-ready")) return;
    root.setAttribute("data-xhc-ready", "1");

    var track = root.querySelector("[data-xhc-track]");
    var prev  = root.querySelector("[data-xhc-prev]");
    var next  = root.querySelector("[data-xhc-next]");
    var total = track ? Math.max(track.children.length, 1) : 1;
    var index = 0;
    var timer = null;

    function inOdooEditor() {
      var c = document.body && document.body.classList;
      return !!c && (c.contains("o_web_editor") || c.contains("o_edit_mode"));
    }

    function move(i) {
      if (!track) return;
      index = (i + total) % total;
      track.style.transform = "translateX(-" + (index * 100) + "%)";
    }

    function restart() {
      if (timer) window.clearInterval(timer);
      if (total <= 1) return;
      timer = window.setInterval(function () {
        if (!inOdooEditor()) move(index + 1);
      }, 4200);
    }

    if (prev) prev.addEventListener("click", function () { move(index - 1); restart(); });
    if (next) next.addEventListener("click", function () { move(index + 1); restart(); });

    restart();
  });
})();
</script>
